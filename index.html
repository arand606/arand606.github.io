local function tween(obj, props, t, style, dir)
    return TweenService:Create(
        obj,
        TweenInfo.new(t or 0.3, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
end

local function pulseViz(isPlaying)
    for i, bar in ipairs(vizBars) do
        local targetH = isPlaying and math.random(15, 100) / 100 or 0.08
        local delay   = (i - 1) * 0.03
        local tw = tween(bar, {Size = UDim2.new(bar.Size.X.Scale, 0, targetH, 0)}, 0.35 + math.random() * 0.25, Enum.EasingStyle.Sine)
        task.delay(delay, function() tw:Play() end)
    end
end
