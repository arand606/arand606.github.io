    return TweenService:Create(
        obj,
        TweenInfo.new(t or 0.3, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),

local index, loadToken, isScrubbing = 1, 0, false
local sound = SoundService:FindFirstChild("MainMusic") or Instance.new("Sound", SoundService)
sound.Name   = "MainMusic"
sound.Volume = 0.5
sound.Looped = true

local function formatTime(s)
    if not s or s ~= s or s == math.huge then return "0:00" end
    return string.format("%d:%02d", math.floor(s / 60), math.floor(s % 60))
end

local vizToken = 0
local function startVizLoop()
    vizToken += 1
    local myToken = vizToken
    task.spawn(function()
        while vizToken == myToken and sound.IsPlaying do
            pulseViz(true)
            task.wait(0.45)
        end
        if vizToken == myToken then
            pulseViz(false)
        end
    end)
end

local function stopViz()
    vizToken += 1
    pulseViz(false)
end

local function flashPanel()
    local tw1 = tween(mainCanvas, {GroupTransparency = 0.35}, 0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    local tw2 = tween(mainCanvas, {GroupTransparency = 0}, 0.3)
    tw1:Play()
    tw1.Completed:Connect(function() tw2:Play() end)
end

local function updateDisplay(loading)
    if loading then
        songNameLabel.Text  = "LOADING..."
        songNameLabel2.Text = "LOADING..."
        indexBadge.Text    = string.format("%02d", index)
        stopViz()
    else
        songNameLabel.Text  = playlist[index].name:upper()
        songNameLabel2.Text = playlist[index].name:upper()
        indexBadge.Text    = string.format("%02d", index)
        timeTotal.Text     = formatTime(sound.TimeLength)
        btnPlay.Text       = sound.IsPlaying and "PAUSE" or "PLAY"
        if sound.IsPlaying then
            startVizLoop()
        else
            stopViz()
        end
    end
    songNameLabel.Position  = UDim2.new(0, 0, 0, 0)
    songNameLabel2.Position = UDim2.new(2, 0, 0, 0)
end

local function loadTrack(autoplay)
    loadToken += 1
    local token = loadToken
    flashPanel()
    updateDisplay(true)
    sound:Stop()
    task.spawn(function()
        sound.SoundId = "rbxassetid://" .. playlist[index].id
        task.wait()
        local timeout = 10
        local elapsed = 0
        while not sound.IsLoaded and elapsed < timeout do
            task.wait(0.1)
            elapsed += 0.1
        end
        if token ~= loadToken then return end
        
        if autoplay then
            sound:Play()
        end
        updateDisplay(false)
    end)
end


btnPlay.MouseButton1Click:Connect(function()
    if sound.IsPlaying then
        sound:Pause()
        stopViz()
    else
        if sound.TimePosition > 0 then
            sound:Resume()
        else
            sound:Play()
        end
        startVizLoop()
    end
    btnPlay.Text = sound.IsPlaying and "PAUSE" or "PLAY"
end)

RunService.RenderStepped:Connect(function()
    if not isScrubbing and sound.IsPlaying and sound.TimeLength > 0 then
        local p = math.clamp(sound.TimePosition / sound.TimeLength, 0, 1)
        fill.Size = UDim2.new(p, 0, 1, 0)
        scrubKnob.Position = UDim2.new(p, 0, 0.5, 0)
        timeElapsed.Text = formatTime(sound.TimePosition)
    end

    local textW = songNameLabel.TextBounds.X
    local clipW = nameClipper.AbsoluteSize.X
    if textW > clipW then
        local GAP    = 50
        local cycle  = textW + GAP
        local speed  = 38
        local offset = (tick() * speed) % cycle

        songNameLabel.TextXAlignment  = Enum.TextXAlignment.Left
        songNameLabel2.TextXAlignment = Enum.TextXAlignment.Left
        songNameLabel.Size    = UDim2.new(0, textW, 1, 0)
        songNameLabel2.Size   = UDim2.new(0, textW, 1, 0)

        songNameLabel.Position  = UDim2.new(0, -offset, 0, 0)
        songNameLabel2.Position = UDim2.new(0, cycle - offset, 0, 0)
    else
        songNameLabel.TextXAlignment  = Enum.TextXAlignment.Center
        songNameLabel2.TextXAlignment = Enum.TextXAlignment.Center
        songNameLabel.Size    = UDim2.new(1, 0, 1, 0)
        songNameLabel2.Size   = UDim2.new(0, 0, 1, 0)
        songNameLabel.Position  = UDim2.new(0, 0, 0, 0)
        songNameLabel2.Position = UDim2.new(2, 0, 0, 0)
    end
end)

barBg.InputBegan:Connect(function(input)
    local t = input.UserInputType
    if t ~= Enum.UserInputType.MouseButton1 and t ~= Enum.UserInputType.Touch then return end
    isScrubbing = true

    local function getScrubPos()
        local mx = UserInputService:GetMouseLocation().X
        return math.clamp((mx - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
    end
    local function applyPos()
        local p = getScrubPos()
        fill.Size = UDim2.new(p, 0, 1, 0)
        scrubKnob.Position = UDim2.new(p, 0, 0.5, 0)
        timeElapsed.Text = formatTime(p * sound.TimeLength)
        return p
    end

    local conn1 = UserInputService.InputChanged:Connect(function(move)
        if move.UserInputType == Enum.UserInputType.MouseMovement
            or move.UserInputType == Enum.UserInputType.Touch then
            applyPos()
        end
    end)
    local conn2
    conn2 = UserInputService.InputEnded:Connect(function(ended)
        if ended.UserInputType == Enum.UserInputType.MouseButton1
            or ended.UserInputType == Enum.UserInputType.Touch then
            sound.TimePosition = sound.TimeLength * applyPos()
            isScrubbing = false
            conn1:Disconnect()
            conn2:Disconnect()
        end
    end)
end)
